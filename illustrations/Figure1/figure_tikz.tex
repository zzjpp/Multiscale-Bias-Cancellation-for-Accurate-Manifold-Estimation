\definecolor{nctrue}{HTML}{30363B}
\definecolor{ncfine}{HTML}{6E8CA7}
\definecolor{nccoarse}{HTML}{B9897C}
\definecolor{nccorrected}{HTML}{318571}
\definecolor{ncnormal}{HTML}{A3AAB2}
\def\ncA{0.135}
\def\ncDOne{0.90}
\def\ncDTwo{1.80}
\def\ncTilt{0.40}
\def\ncWTwo{-0.80}
\pgfmathdeclarefunction{ncf}{1}{\pgfmathparse{\ncA*(#1)^2}}
\pgfmathdeclarefunction{ncb}{1}{\pgfmathparse{1-2*\ncA*\ncTilt*(#1)}}
% Vertical displacement along (ncTilt,1), from the fine to the coarse curve.
\pgfmathdeclarefunction{ncs}{1}{\pgfmathparse{
  2*(\ncDTwo-\ncDOne)/(ncb(#1)+sqrt(ncb(#1)^2
  -4*\ncA*\ncTilt^2*(\ncDTwo-\ncDOne)))}}
% Same affine-extrapolated curve, reparameterized by its displayed x coordinate.
\pgfmathdeclarefunction{ncsc}{1}{\pgfmathparse{
  2*(\ncDTwo-\ncDOne)/(ncb(#1)+sqrt(ncb(#1)^2
  -4*\ncA*\ncTilt^2*(1-2*\ncWTwo)*(\ncDTwo-\ncDOne)))}}
\pgfmathdeclarefunction{nccy}{1}{\pgfmathparse{
  ncf((#1)-\ncWTwo*\ncTilt*ncsc(#1))+\ncDOne+\ncWTwo*ncsc(#1)}}
\pgfmathsetmacro{\ncSZero}{ncs(0)}
\pgfmathsetmacro{\ncZX}{\ncTilt*\ncSZero}
\pgfmathsetmacro{\ncZY}{\ncDOne+\ncSZero}
\pgfmathsetmacro{\ncCX}{\ncWTwo*\ncTilt*\ncSZero}
\pgfmathsetmacro{\ncCY}{\ncDOne+\ncWTwo*\ncSZero}
\pgfmathsetmacro{\ncOracleC}{(1-\ncWTwo)*\ncDOne+\ncWTwo*\ncDTwo}
\begin{tikzpicture}[

  curve corrected/.style={draw=nccorrected,line width=1.10pt,
    dash pattern=on 7.0pt off 2.0pt on .9pt off 2.0pt},
  true normal/.style={draw=ncnormal,line width=.65pt},
  estimated normal/.style={draw=nctrue,line width=.80pt,dash pattern=on 4.0pt off 2.3pt},
  oracle point/.style={circle,draw=nccoarse,fill=white,line width=.75pt,
    inner sep=0pt,minimum size=3.6pt,outer sep=0pt},
  fine point/.style={circle,draw=ncfine,fill=white,line width=.75pt,
    inner sep=0pt,minimum size=3.6pt,outer sep=0pt},
  estimated point/.style={rectangle,draw=nccoarse,fill=white,line width=.75pt,
    inner sep=0pt,minimum size=3.5pt,outer sep=0pt},
  corrected point/.style={diamond,draw=nccorrected,fill=nccorrected,line width=.6pt,
